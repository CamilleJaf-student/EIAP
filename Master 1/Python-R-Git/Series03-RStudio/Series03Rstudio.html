7 We need to see if we have a interaction between ID and GROUP

anova_results <- ezANOVA(data = results_clean2 ,
                         dv = .(MovementTime) ,
                         wid = .(SUBJ),
                         within = .(ID),
                         within_full = .(ID, TRIAL, ORI, DIR, REP),
                         between = .(GROUP),
                         detailed = FALSE ,
                         type = 3)
